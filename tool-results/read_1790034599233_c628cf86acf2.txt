   119→const CA_MAX = 30
   120→const EXAM_MAX = 70

   122→function liveTotal(ca: string, exam: string): number | null {
   123→  const c = ca === '' ? null : Number(ca)
   124→  const e = exam === '' ? null : Number(exam)
   125→  if (c == null || e == null || !Number.isFinite(c) || !Number.isFinite(e)) return null
   126→  return c + e
   127→}
   128→
   129→function liveGrade(total: number | null, boundaries: GradeBoundary[]): string | null {
   130→  if (total == null || !Number.isFinite(total)) return null
   131→  for (const b of boundaries) {
   132→    if (total >= b.min && total <= b.max) return b.grade
   133→  }
   134→  return boundaries[boundaries.length - 1]?.grade ?? 'F'
   135→}
   136→
   137→function validateInput(ca: string, exam: string): string | null {
   138→  if (ca === '' || exam === '') return null // empty is OK (not yet entered)
   139→  const c = Number(ca)
   140→  const e = Number(exam)
   141→  if (!Number.isFinite(c) || !Number.isFinite(e)) return 'Scores must be valid numbers'
   142→  if (c < 0 || c > CA_MAX) return `CA must be between 0 and ${CA_MAX}`
   143→  if (e < 0 || e > EXAM_MAX) return `Examination must be between 0 and ${EXAM_MAX}`
   144→  return null
   145→}
   146→
   147→function liveCumulative(
   148→  termOrder: number,
   149→  termTotal: number | null,
   150→  priors: { firstTerm: number | null; secondTerm: number | null },
   151→): number | null {
   152→  if (termTotal == null || !Number.isFinite(termTotal)) return null
   153→  if (termOrder === 1) return termTotal
   154→  if (termOrder === 2) {
   155→    if (priors.firstTerm == null) return null
   156→    return Math.round(((priors.firstTerm + termTotal) / 2) * 100) / 100
   157→  }
   158→  if (termOrder === 3) {
   159→    if (priors.firstTerm == null || priors.secondTerm == null) return null
   160→    return (
   161→      Math.round(((priors.firstTerm + priors.secondTerm + termTotal) / 3) * 100) / 100
   162→    )
   163→  }
   164→  return null
   165→}

   179→        <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-transparent">

   185→    default:
   186→      return <Badge variant="outline">{status}</Badge>
   187→  }

   190→function ordinal(n: number | null): string {
   191→  if (!n || n <= 0) return '—'
   192→  const s = ['th', 'st', 'nd', 'rd']
   193→  const v = n % 100
   194→  return n + (s[(v - 20) % 10] || s[v] || s[0])
   195→}

   273→export function TeacherResultsEntry() {

   275→  const [assignmentId, setAssignmentId] = useState<string>('')
   276→
   277→  // Bootstrap — assignments, active session/term, remarks, grade boundaries
   278→  const { data: boot, isLoading: bootLoading } = useQuery({
   279→    queryKey: ['results-bootstrap'],
   280→    queryFn: fetchBootstrap,
   281→  })
   282→
   283→  const activeSession = boot?.activeSession ?? null
   284→  const activeTerm = boot?.activeTerm ?? null
   285→  const assignments = boot?.assignments ?? []
   286→  const remarks = boot?.remarks ?? []
   287→  const boundaries = boot?.gradeBoundaries ?? []
   288→
   289→  const selectedAssignment = useMemo(
   290→    () => assignments.find((a) => a.id === assignmentId) ?? null,
   291→    [assignments, assignmentId],
   292→  )
   293→
   294→  // Students list
   295→  const studentsQuery = useQuery({
   296→    queryKey: ['results-students', selectedAssignment?.classArmId],
   297→    queryFn: () => fetchStudents(selectedAssignment!.classArmId),
   298→    enabled: !!selectedAssignment?.classArmId && !!activeSession && !!activeTerm,
   299→  })
   300→
   301→  // Existing results for this session/term/arm/subject
   302→  const resultsQuery = useQuery({
   303→    queryKey: [
   304→      'results',
   305→      activeSession?.id,
   306→      activeTerm?.id,
   307→      selectedAssignment?.classArmId,
   308→      selectedAssignment?.subjectId,
   309→    ],
   310→    queryFn: () =>
   311→      fetchResults({
   312→        sessionId: activeSession!.id,
   313→        termId: activeTerm!.id,
   314→        classArmId: selectedAssignment!.classArmId,
   315→        subjectId: selectedAssignment!.subjectId,
   316→      }),
   317→    enabled:
   318→      !!selectedAssignment?.classArmId &&
   319→      !!selectedAssignment?.subjectId &&
   320→      !!activeSession &&
   321→      !!activeTerm,
   322→  })
   323→
   324→  const students = studentsQuery.data ?? []
   325→  const results = resultsQuery.data ?? []
   326→
   327→  // Build per-student row state from students + results
   328→  const rowStates = useMemo<RowState[]>(() => {
   329→    return students.map((s) => {
   330→      const r = results.find((rr) => rr.studentId === s.id)
   331→      return {
   332→        studentId: s.id,
   333→        ca: r?.ca != null ? String(r.ca) : '',
   334→        exam: r?.exam != null ? String(r.exam) : '',
   335→        remarkId: r?.remarkId ?? null,
   336→        serverCa: r?.ca ?? null,
   337→        serverExam: r?.exam ?? null,
   338→        serverRemarkId: r?.remarkId ?? null,
   339→        status: r?.status ?? 'NEW',
   340→        position: r?.position ?? null,
   341→        priorTotals: r?.priorTotals ?? { firstTerm: null, secondTerm: null },
   342→        resultId: r?.id ?? null,
   343→      }
   344→    })
   345→  }, [students, results])
   346→
   347→  // Per-row draft overrides (so user can edit before saving)
   348→  const [drafts, setDrafts] = useState<Record<string, { ca: string; exam: string; remarkId: string | null }>>({})
   349→
   350→  // Reset drafts whenever the underlying data changes
   351→  // (use a key based on result snapshot signature to reset cleanly)
   352→  const draftSignature = rowStates
   353→    .map((r) => `${r.studentId}:${r.serverCa ?? ''}:${r.serverExam ?? ''}:${r.serverRemarkId ?? ''}`)
   354→    .join('|')
   355→  const [lastSig, setLastSig] = useState('')
   356→  if (draftSignature !== lastSig) {
   357→    setLastSig(draftSignature)
   358→    setDrafts({})
   359→  }
   360→
   361→  function getDraft(studentId: string, fallback: RowState) {
   362→    const d = drafts[studentId]
   363→    if (d) return d
   364→    return { ca: fallback.ca, exam: fallback.exam, remarkId: fallback.remarkId }
   365→  }
   366→
   367→  function setDraft(studentId: string, patch: Partial<{ ca: string; exam: string; remarkId: string | null }>) {
   368→    setDrafts((prev) => {
   369→      const existing = prev[studentId] ?? {
   370→        ca: rowStates.find((r) => r.studentId === studentId)?.ca ?? '',
   371→        exam: rowStates.find((r) => r.studentId === studentId)?.exam ?? '',
   372→        remarkId: rowStates.find((r) => r.studentId === studentId)?.remarkId ?? null,
   373→      }
   374→      return { ...prev, [studentId]: { ...existing, ...patch } }
   375→    })
   376→  }
   377→
   378→  // Save mutation (single row)
   379→  const saveMutation = useMutation({
   380→    mutationFn: (vars: {
   381→      row: RowState
   382→      ca: string
   383→      exam: string
   384→      remarkId: string | null
   385→    }) => {
   386→      if (!selectedAssignment || !activeSession || !activeTerm) {
   387→        throw new Error('Missing assignment, session, or term')
   388→      }
   389→      const ca = Number(vars.ca)
   390→      const exam = Number(vars.exam)
   391→      const body: SaveBody = {
   392→        studentId: vars.row.studentId,
   393→        subjectId: selectedAssignment.subjectId,
   394→        sessionId: activeSession.id,
   395→        termId: activeTerm.id,
   396→        classArmId: selectedAssignment.classArmId,
   397→        ca,
   398→        exam,
   399→        remarkId: vars.remarkId,
   400→      }
   401→      return saveResult(body)
   402→    },
   403→    onSuccess: () => {
   404→      toast.success('Result saved')
   405→      qc.invalidateQueries({
   406→        queryKey: ['results'],
   407→      })
   408→    },
   409→    onError: (err: unknown) => {
   410→      const msg =
   411→        err instanceof ApiError
   412→          ? err.message
   413→          : err instanceof Error
   414→            ? err.message
   415→            : 'Failed to save'
   416→      toast.error(msg)
   417→    },
   418→  })
   419→
   420→  // Submit mutation (batch — either all drafts OR by studentIds)
   421→  const submitMutation = useMutation({
   422→    mutationFn: (vars: { studentIds: string[]; subjectId: string; classArmId: string; sessionId: string; termId: string }) => {
   423→      return submitBatch(vars)
   424→    },
   425→    onSuccess: (data) => {
   426→      if (data.submittedCount > 0) {
   427→        toast.success(`${data.submittedCount} result(s) submitted`)
   428→      }
   429→      if (data.rejectedCount > 0) {
   430→        toast.warning(`${data.rejectedCount} result(s) could not be submitted`)
   431→      }
   432→      qc.invalidateQueries({ queryKey: ['results'] })
   433→    },
   434→    onError: (err: unknown) => {
   435→      const msg =
   436→        err instanceof ApiError
   437→          ? err.message
   438→          : err instanceof Error
   439→            ? err.message
   440→            : 'Failed to submit'
   441→      toast.error(msg)
   442→    },
   443→  })

   445→  function handleSaveRow(row: RowState) {
   446→    const draft = getDraft(row.studentId, row)
   447→    const err = validateInput(draft.ca, draft.exam)
   448→    if (err) return toast.error(err)
   449→    if (draft.ca === '' || draft.exam === '') {
   450→      return toast.error('Enter both CA and Examination scores')
   451→    }
   452→    saveMutation.mutate({
   453→      row,
   454→      ca: draft.ca,
   455→      exam: draft.exam,
   456→      remarkId: draft.remarkId,
   457→    })
   458→  }

   460→  function handleSaveAll() {
   461→    const editable = rowStates.filter((r) => r.status !== 'SUBMITTED' && r.status !== 'APPROVED')
   462→    if (editable.length === 0) return toast.info('Nothing to save')
   463→
   464→    // Validate all
   465→    for (const r of editable) {
   466→      const d = getDraft(r.studentId, r)
   467→      const err = validateInput(d.ca, d.exam)
   468→      if (err) return toast.error(`Row error: ${err}`)
   469→      if (d.ca === '' || d.exam === '') {
   470→        return toast.error('Enter both CA and Examination for every student')
   471→      }
   472→    }
   473→    // Save one by one (awaitable)
   474→    Promise.all(
   475→      editable.map((r) => {
   476→        const d = getDraft(r.studentId, r)
   477→        return saveMutation.mutateAsync({ row: r, ca: d.ca, exam: d.exam, remarkId: d.remarkId })
   478→      }),
   479→    )
   480→      .then(() => toast.success(`Saved ${editable.length} result(s)`))
   481→      .catch((e) => {
   482→        const msg = e instanceof ApiError ? e.message : 'Some saves failed'
   483→        toast.error(msg)
   484→      })
   485→  }
