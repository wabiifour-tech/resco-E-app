   142→function studentName(r: ResultRow): string {

   146→function ordinal(n: number | null): string {
   147→  if (!n || n <= 0) return '—'
   148→  const s = ['th', 'st', 'nd', 'rd']
   149→  const v = n % 100
   150→  return n + (s[(v - 20) % 10] || s[v] || s[0])
   151→}

   184→// Group results by (classArmId + subjectId) so the principal can review a
   185→// whole subject-class "result sheet" at once.
   186→function groupResults(rows: ResultRow[]) {
   187→  const map = new Map<
   188→    string,
   189→    {
   190→      key: string
   191→      classArmId: string
   192→      classArmName: string
   193→      className: string
   194→      subjectId: string
   195→      subjectName: string
   196→      enteredByTeacherName: string | null
   197→      rows: ResultRow[]
   198→    }
   199→  >()
   200→  for (const r of rows) {
   201→    const k = `${r.classArmId}|${r.subjectId}`
   202→    if (!map.has(k)) {
   203→      map.set(k, {
   204→        key: k,
   205→        classArmId: r.classArm.id,
   206→        classArmName: r.classArm.fullName,
   207→        className: r.classArm.class.name,
   208→        subjectId: r.subject.id,
   209→        subjectName: r.subject.name,
   210→        enteredByTeacherName: r.enteredByTeacherName ?? null,
   211→        rows: [],
   212→      })
   213→    }
   214→    map.get(k)!.rows.push(r)
   215→  }
   216→  return Array.from(map.values())
   217→}

   278→function ReturnFormBody({
   279→  result,
   280→  onOpenChange,
   281→}: {
   282→  result: ResultRow
   283→  onOpenChange: (o: boolean) => void
   284→}) {
   285→  const qc = useQueryClient()
   286→  const [reason, setReason] = useState('')
   287→
   288→  const mutation = useMutation({
   289→    mutationFn: (r: string) => returnResult(result.id, r),
   290→    onSuccess: () => {
   291→      qc.invalidateQueries({ queryKey: ['approvals'] })
   292→      toast.success(
   293→        `Result returned for correction: ${studentName(result)}`,
   294→      )
   295→      onOpenChange(false)
   296→    },
   297→    onError: (err: unknown) => {
   298→      const msg =
   299→        err instanceof ApiError
   300→          ? err.message
   301→          : err instanceof Error
   302→            ? err.message
   303→            : 'Failed to return result'
   304→      toast.error(msg)
   305→    },
   306→  })
   307→
   308→  function handleSubmit(e: React.FormEvent) {
   309→    e.preventDefault()
   310→    if (!reason.trim()) return toast.error('A reason is required')
   311→    mutation.mutate(reason.trim())
   312→  }
   313→
   314→  return (
   315→    <>
   316→      <DialogHeader>
   317→        <DialogTitle className="flex items-center gap-2">
   318→          <Undo2 className="h-4 w-4" />
   319→          Return for correction
   320→        </DialogTitle>
   321→        <DialogDescription>
   322→          Send this result back to{' '}
   323→          <span className="font-medium text-foreground">
   324→            {result.enteredByTeacherName ?? 'the teacher'}
   325→          </span>{' '}
   326→          for correction. The student will be hidden from the approvals queue
   327→          until the teacher re-submits.
   328→        </DialogDescription>
   329→      </DialogHeader>
   330→
   331→      <div className="rounded-md border p-3 text-sm space-y-1">
   332→        <div className="flex justify-between gap-2">
   333→          <span className="text-muted-foreground">Student</span>
   334→          <span className="font-medium text-right">
   335→            {studentName(result)}
   336→            <span className="block text-xs text-muted-foreground font-normal">
   337→              {result.student.admissionNumber}
   338→            </span>
   339→          </span>
   340→        </div>
   341→        <div className="flex justify-between gap-2">
   342→          <span className="text-muted-foreground">Subject</span>
   343→          <span className="font-medium text-right">{result.subject.name}</span>
   344→        </div>
   345→        <div className="flex justify-between gap-2">
   346→          <span className="text-muted-foreground">Class</span>
   347→          <span className="font-medium text-right">
   348→            {result.classArm.fullName}
   349→          </span>
   350→        </div>
   351→        <div className="flex justify-between gap-2">
   352→          <span className="text-muted-foreground">Total</span>
   353→          <span className="font-medium text-right">
   354→            {result.total ?? '—'} / 100 ({result.grade ?? '—'})
   355→          </span>
   356→        </div>
   357→      </div>
   358→
   359→      <form onSubmit={handleSubmit} className="space-y-3">
   360→        <div className="space-y-2">
   361→          <Label htmlFor="reason">Reason / correction note</Label>
   362→          <Textarea
   363→            id="reason"
   364→            value={reason}
   365→            onChange={(e) => setReason(e.target.value)}
   366→            placeholder="e.g. CA total should be 28 not 38. Exam score appears transposed. Please verify and re-submit."
   367→            rows={4}
   368→            maxLength={500}
   369→            autoFocus
   370→            required
   371→          />
   372→          <p className="text-xs text-muted-foreground">
   373→            The teacher will see this note. {reason.length}/500 characters.
   374→          </p>
   375→        </div>
   376→
   377→        <DialogFooter>
   378→          <Button
   379→            type="button"
   380→            variant="outline"
   381→            onClick={() => onOpenChange(false)}
   382→            disabled={mutation.isPending}
   383→          >
   384→            Cancel
   385→          </Button>
   386→          <Button type="submit" variant="destructive" disabled={mutation.isPending}>
   387→            {mutation.isPending && (
   388→              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   389→            )}
   390→            Return for correction
   391→          </Button>
   392→        </DialogFooter>
   393→      </form>
   394→    </>
   395→  )
   396→}

   400→function ReopenDialog({
   401→  open,
   402→  onOpenChange,
   403→  result,
   404→}: {
   405→  open: boolean
   406→  onOpenChange: (o: boolean) => void
   407→  result: ResultRow | null
   408→}) {
   409→  const qc = useQueryClient()
   410→  const mutation = useMutation({
   411→    mutationFn: (id: string) => unlockResult(id),
   412→    onSuccess: () => {
   413→      qc.invalidateQueries({ queryKey: ['approvals'] })
   414→      toast.success('Result reopened — the teacher can now edit it again.')
   415→      onOpenChange(false)
   416→    },
   417→    onError: (err: unknown) => {
   418→      const msg =
   419→        err instanceof ApiError
   420→          ? err.message
   421→          : err instanceof Error
   422→            ? err.message
   423→            : 'Failed to reopen result'
   424→      toast.error(msg)
   425→    },
   426→  })
   427→
   428→  return (
   429→    <AlertDialog
   430→      open={open && !!result}
   431→      onOpenChange={(o) => {
   432→        if (!mutation.isPending) onOpenChange(o)
   433→      }}
   434→    >
   435→      <AlertDialogContent>
   436→        <AlertDialogHeader>
   437→          <AlertDialogTitle className="flex items-center gap-2">
   438→            <Unlock className="h-4 w-4" /> Reopen this result?
   439→          </AlertDialogTitle>
   440→          <AlertDialogDescription>
   441→            {result ? (
   442→              <>
   443→                <span className="font-medium text-foreground">
   444→                  {studentName(result)}
   445→                </span>{' '}
   446→                — {result.subject.name} ({result.classArm.fullName}) will be
   447→                unlocked and set back to <span className="font-medium">Pending</span>.
   448→                The teacher will be able to edit and re-submit it. Approved
   449→                audit history is kept.
   450→              </>
   451→            ) : null}
   452→          </AlertDialogDescription>
   453→        </AlertDialogHeader>
   454→        <AlertDialogFooter>
   455→          <AlertDialogCancel disabled={mutation.isPending}>
   456→            Cancel
   457→          </AlertDialogCancel>
   458→          <AlertDialogAction
   459→            disabled={mutation.isPending}
   460→            onClick={(e) => {
   461→              e.preventDefault()
   462→              if (result) mutation.mutate(result.id)
   463→            }}
   464→          >
   465→            {mutation.isPending && (
   466→              <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   467→            )}
   468→            Reopen for editing
   469→          </AlertDialogAction>
   470→        </AlertDialogFooter>
   471→      </AlertDialogContent>
   472→    </AlertDialog>
   473→  )
   474→}

   611→function ResultSheetGroup({
   612→  group,
   613→  selected,
   614→  onToggleSelect,
   615→  onToggleGroup,
   616→  onApproveOne,
   617→  onApproveBatch,
   618→  onReturn,
   619→  onReopen,
   620→  busyIds,
   621→}: {
   622→  group: ReturnType<typeof groupResults>[number]
   623→  selected: Set<string>
   624→  onToggleSelect: (id: string, checked: boolean) => void
   625→  onToggleGroup: (group: ReturnType<typeof groupResults>[number], checked: boolean) => void
   626→  onApproveOne: (id: string) => void
   627→  onApproveBatch: (ids: string[]) => void
   628→  onReturn: (r: ResultRow) => void
   629→  onReopen: (r: ResultRow) => void
   630→  busyIds: Set<string>
   631→}) {
   632→  const submittedRows = group.rows.filter((r) => r.status === 'SUBMITTED')
   633→  const selectedCount = submittedRows.filter((r) => selected.has(r.id)).length
   634→  const allSelected =
   635→    submittedRows.length > 0 && selectedCount === submittedRows.length
   636→  const someSelected = selectedCount > 0 && !allSelected
   637→  const hasApproved = group.rows.some((r) => r.status === 'APPROVED')
   638→  const hasNeedsCorrection = group.rows.some(
   639→    (r) => r.status === 'NEEDS_CORRECTION',
   640→  )
   641→
