   117→const CA_MAX = 30
   118→const EXAM_MAX = 70
   119→
   120→function liveTotal(ca: string, exam: string): number | null {
   121→  const c = ca === '' ? null : Number(ca)
   122→  const e = exam === '' ? null : Number(exam)
   123→  if (c == null || e == null || !Number.isFinite(c) || !Number.isFinite(e)) return null
   124→  return c + e
   125→}
   126→
   127→function liveGrade(total: number | null, boundaries: GradeBoundary[]): string | null {
   128→  if (total == null || !Number.isFinite(total)) return null
   129→  for (const b of boundaries) {
   130→    if (total >= b.min && total <= b.max) return b.grade
   131→  }
   132→  return boundaries[boundaries.length - 1]?.grade ?? 'F'
   133→}
   134→
   135→function validateInput(ca: string, exam: string): string | null {
   136→  if (ca === '' || exam === '') return null // empty is OK (not yet entered)
   137→  const c = Number(ca)
   138→  const e = Number(exam)
   139→  if (!Number.isFinite(c) || !Number.isFinite(e)) return 'Scores must be valid numbers'
   140→  if (c < 0 || c > CA_MAX) return `CA must be between 0 and ${CA_MAX}`
   141→  if (e < 0 || e > EXAM_MAX) return `Examination must be between 0 and ${EXAM_MAX}`
   142→  return null
   143→}
   144→
   145→function liveCumulative(
   146→  termOrder: number,
   147→  termTotal: number | null,
   148→  priors: { firstTerm: number | null; secondTerm: number | null },
   149→): number | null {
   150→  if (termTotal == null || !Number.isFinite(termTotal)) return null
   151→  if (termOrder === 1) return termTotal
   152→  if (termOrder === 2) {
   153→    if (priors.firstTerm == null) return null
   154→    return Math.round(((priors.firstTerm + termTotal) / 2) * 100) / 100
   155→  }
   156→  if (termOrder === 3) {
   157→    if (priors.firstTerm == null || priors.secondTerm == null) return null
   158→    return (
   159→      Math.round(((priors.firstTerm + priors.secondTerm + termTotal) / 3) * 100) / 100
   160→    )
   161→  }
   162→  return null
   163→}

   177→        <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 border-transparent">

   183→    default:
   184→      return <Badge variant="outline">{status}</Badge>
   185→  }

   188→function ordinal(n: number | null): string {
   189→  if (!n || n <= 0) return '—'
   190→  const s = ['th', 'st', 'nd', 'rd']
   191→  const v = n % 100
   192→  return n + (s[(v - 20) % 10] || s[v] || s[0])
   193→}

   271→export function TeacherResultsEntry() {
   272→  const qc = useQueryClient()

   274→
   275→  // Bootstrap — assignments, active session/term, remarks, grade boundaries
   276→  const { data: boot, isLoading: bootLoading } = useQuery({
   277→    queryKey: ['results-bootstrap'],
   278→    queryFn: fetchBootstrap,
   279→  })
   280→
   281→  const activeSession = boot?.activeSession ?? null
   282→  const activeTerm = boot?.activeTerm ?? null
   283→  const assignments = boot?.assignments ?? []
   284→  const remarks = boot?.remarks ?? []
   285→  const boundaries = boot?.gradeBoundaries ?? []
   286→
   287→  const selectedAssignment = useMemo(
   288→    () => assignments.find((a) => a.id === assignmentId) ?? null,
   289→    [assignments, assignmentId],
   290→  )
   291→
   292→  // Students list
   293→  const studentsQuery = useQuery({
   294→    queryKey: ['results-students', selectedAssignment?.classId],
   295→    queryFn: () => fetchStudents(selectedAssignment!.classId),
   296→    enabled: !!selectedAssignment?.classId && !!activeSession && !!activeTerm,
   297→  })
   298→
   299→  // Existing results for this session/term/class/subject
   300→  const resultsQuery = useQuery({
   301→    queryKey: [
   302→      'results',
   303→      activeSession?.id,
   304→      activeTerm?.id,
   305→      selectedAssignment?.classId,
   306→      selectedAssignment?.subjectId,
   307→    ],
   308→    queryFn: () =>
   309→      fetchResults({
   310→        sessionId: activeSession!.id,
   311→        termId: activeTerm!.id,
   312→        classId: selectedAssignment!.classId,
   313→        subjectId: selectedAssignment!.subjectId,
   314→      }),
   315→    enabled:
   316→      !!selectedAssignment?.classId &&
   317→      !!selectedAssignment?.subjectId &&
   318→      !!activeSession &&
   319→      !!activeTerm,
   320→  })
   321→
   322→  const students = studentsQuery.data ?? []
   323→  const results = resultsQuery.data ?? []
   324→
   325→  // Build per-student row state from students + results
   326→  const rowStates = useMemo<RowState[]>(() => {
   327→    return students.map((s) => {
   328→      const r = results.find((rr) => rr.studentId === s.id)
   329→      return {
   330→        studentId: s.id,
   331→        ca: r?.ca != null ? String(r.ca) : '',
   332→        exam: r?.exam != null ? String(r.exam) : '',
   333→        remarkId: r?.remarkId ?? null,
   334→        serverCa: r?.ca ?? null,
   335→        serverExam: r?.exam ?? null,
   336→        serverRemarkId: r?.remarkId ?? null,
   337→        status: r?.status ?? 'NEW',
   338→        position: r?.position ?? null,
   339→        priorTotals: r?.priorTotals ?? { firstTerm: null, secondTerm: null },
   340→        resultId: r?.id ?? null,
   341→      }
   342→    })
   343→  }, [students, results])
   344→
   345→  // Per-row draft overrides (so user can edit before saving)
   346→  const [drafts, setDrafts] = useState<Record<string, { ca: string; exam: string; remarkId: string | null }>>({})
   347→
   348→  // Reset drafts whenever the underlying data changes
   349→  // (use a key based on result snapshot signature to reset cleanly)
   350→  const draftSignature = rowStates
   351→    .map((r) => `${r.studentId}:${r.serverCa ?? ''}:${r.serverExam ?? ''}:${r.serverRemarkId ?? ''}`)
   352→    .join('|')
   353→  const [lastSig, setLastSig] = useState('')
   354→  if (draftSignature !== lastSig) {
   355→    setLastSig(draftSignature)
   356→    setDrafts({})
   357→  }
   358→
   359→  function getDraft(studentId: string, fallback: RowState) {
   360→    const d = drafts[studentId]
   361→    if (d) return d
   362→    return { ca: fallback.ca, exam: fallback.exam, remarkId: fallback.remarkId }
   363→  }
   364→
   365→  function setDraft(studentId: string, patch: Partial<{ ca: string; exam: string; remarkId: string | null }>) {
   366→    setDrafts((prev) => {
   367→      const existing = prev[studentId] ?? {
   368→        ca: rowStates.find((r) => r.studentId === studentId)?.ca ?? '',
   369→        exam: rowStates.find((r) => r.studentId === studentId)?.exam ?? '',
   370→        remarkId: rowStates.find((r) => r.studentId === studentId)?.remarkId ?? null,
   371→      }
   372→      return { ...prev, [studentId]: { ...existing, ...patch } }
   373→    })
   374→  }
   375→
   376→  // Save mutation (single row)
   377→  const saveMutation = useMutation({
   378→    mutationFn: (vars: {
   379→      row: RowState
   380→      ca: string
   381→      exam: string
   382→      remarkId: string | null
   383→    }) => {
   384→      if (!selectedAssignment || !activeSession || !activeTerm) {
   385→        throw new Error('Missing assignment, session, or term')
   386→      }
   387→      const ca = Number(vars.ca)
   388→      const exam = Number(vars.exam)
   389→      const body: SaveBody = {
   390→        studentId: vars.row.studentId,
   391→        subjectId: selectedAssignment.subjectId,
   392→        sessionId: activeSession.id,
   393→        termId: activeTerm.id,
   394→        classId: selectedAssignment.classId,
   395→        ca,
   396→        exam,
   397→        remarkId: vars.remarkId,
   398→      }
   399→      return saveResult(body)
   400→    },
   401→    onSuccess: () => {
   402→      toast.success('Result saved')
   403→      qc.invalidateQueries({
   404→        queryKey: ['results'],
   405→      })
   406→    },
   407→    onError: (err: unknown) => {
   408→      const msg =
   409→        err instanceof ApiError
   410→          ? err.message
   411→          : err instanceof Error
   412→            ? err.message
   413→            : 'Failed to save'
   414→      toast.error(msg)
   415→    },
   416→  })
   417→
   418→  // Submit mutation (batch — either all drafts OR by studentIds)
   419→  const submitMutation = useMutation({
   420→    mutationFn: (vars: { studentIds: string[]; subjectId: string; classId: string; sessionId: string; termId: string }) => {
   421→      return submitBatch(vars)
   422→    },
   423→    onSuccess: (data) => {
   424→      if (data.submittedCount > 0) {
   425→        toast.success(`${data.submittedCount} result(s) submitted`)
   426→      }
   427→      if (data.rejectedCount > 0) {
   428→        toast.warning(`${data.rejectedCount} result(s) could not be submitted`)
   429→      }
   430→      qc.invalidateQueries({ queryKey: ['results'] })
   431→    },
   432→    onError: (err: unknown) => {
   433→      const msg =
   434→        err instanceof ApiError
   435→          ? err.message
   436→          : err instanceof Error
   437→            ? err.message
   438→            : 'Failed to submit'
   439→      toast.error(msg)
   440→    },
   441→  })

   443→  function handleSaveRow(row: RowState) {
   444→    const draft = getDraft(row.studentId, row)
   445→    const err = validateInput(draft.ca, draft.exam)
   446→    if (err) return toast.error(err)
   447→    if (draft.ca === '' || draft.exam === '') {
   448→      return toast.error('Enter both CA and Examination scores')
   449→    }
   450→    saveMutation.mutate({
   451→      row,
   452→      ca: draft.ca,
   453→      exam: draft.exam,
   454→      remarkId: draft.remarkId,
   455→    })
   456→  }

   458→  function handleSaveAll() {
   459→    const editable = rowStates.filter((r) => r.status !== 'SUBMITTED' && r.status !== 'APPROVED')
   460→    if (editable.length === 0) return toast.info('Nothing to save')
   461→
   462→    // Validate all
   463→    for (const r of editable) {
   464→      const d = getDraft(r.studentId, r)
   465→      const err = validateInput(d.ca, d.exam)
   466→      if (err) return toast.error(`Row error: ${err}`)
   467→      if (d.ca === '' || d.exam === '') {
   468→        return toast.error('Enter both CA and Examination for every student')
   469→      }
   470→    }
   471→    // Save one by one (awaitable)
   472→    Promise.all(
   473→      editable.map((r) => {
   474→        const d = getDraft(r.studentId, r)
   475→        return saveMutation.mutateAsync({ row: r, ca: d.ca, exam: d.exam, remarkId: d.remarkId })
   476→      }),
   477→    )
   478→      .then(() => toast.success(`Saved ${editable.length} result(s)`))
   479→      .catch((e) => {
   480→        const msg = e instanceof ApiError ? e.message : 'Some saves failed'
   481→        toast.error(msg)
   482→      })
   483→  }
